\section{Singular Value Decomposition - SVD}

\subsection{Definizione}
\begin{frame}{Fattorizzazione SVD}
\begin{definition}
Sia $A\in\mathbb{R}^{m\times n}$, oppure  $A\in\mathbb{C}^{m\times n}$, e \( Rg(A)=r\) allora la decomposizione ai valori singolari di una matrice è definita come:
\[
A = U \Sigma V^\top      
\]
Dove: 
\begin{itemize}
    \item \(U \in \R^{m \times m}\) è la matrice contenente i \textit{vettori singolari di sinistra} di \(A\), ovvero gli autovalori di \(A A^\top\)
    \item \( \Sigma \in \R^{m \times n} \) è la matrice contenente i \textit{valori singolari} di \(A\), ovvero le radici quadrate degli autovalori in comune di \(AA^\top\) e \(A^\top A\)
    \item \( V \in \R^{n \times n} \) è la matrice composta dagli autovettori di \( A^\top A\) detti anche \textit{vettori singolari di destra}
\end{itemize}
\end{definition}
\end{frame}

\begin{frame}
\[
A = 
\begin{bmatrix}
| & | & & | \\
\mathbf{u}_1 & \mathbf{u}_2 & \cdots & \mathbf{u}_m \\
| & | & & |
\end{bmatrix} 
\begin{bmatrix}
\sigma_1 & 0 & \cdots & 0 & \cdots & 0 \\
0 & \sigma_2 & \cdots & 0 & \cdots & 0 \\
\vdots & \vdots & \ddots & \vdots  & \ddots & \vdots \\
0 & 0 & \cdots & \sigma_r & \cdots & 0 \\
0 & 0 & \cdots & 0 & \cdots & 0 \\
\vdots & \vdots & \ddots & \vdots & \ddots & \vdots \\
0 & 0 & \cdots & 0 & \cdots & 0
\end{bmatrix} \begin{bmatrix}
| & | & & | \\
\mathbf{v}_1 & \mathbf{v}_2 & \cdots & \mathbf{v}_n \\
| & | & & |
\end{bmatrix} ^\top      
\]
\end{frame}

\subsection{Applicazioni}
\begin{frame}
\begin{proposition}
\textbf{Fattorizzazione SVD e matrice pseudoinversa}\\
\textit{Hp:} Sia $A \in \R^{m \times n}$ e $A=U\Sigma V^T$ la sua decomposizione a valori singolari\\
\textit{Ts:} La matrice pseudoinversa di $A$ è $A^+=V\Sigma^+U^T \ \in \R^{m \times n}$ dove i valori singolari di $A^+$ sono i reciproci dei valori singolari di $A$ 
\end{proposition}
\end{frame}

\begin{frame}
\begin{proposition}
\textbf{Soluzione ai Minimi Quadrati}\\
\textit{Hp:} Sia $A \in \mathbb{R}^{m \times n}$ una matrice di rango massimo con $m \geq n$, e sia $\mathbf{b} \in \mathbb{R}^m$ e \( A= U' \Sigma' V'^\top \) la fattorizzazione SVD ridotta\\
\textit{Ts:} Per il sistema di equazioni lineari sovradeterminato $A\mathbf{x} = \mathbf{b}$, la soluzione ai minimi quadrati $\mathbf{x}^*$ risolve il seguente problema:
\[
\mathbf{x^*} = \underset{\mathbf{x} \in \mathbb{R}^n}{\mathrm{arg\,min}} \, \| A\mathbf{x} - \mathbf{b} \|_2
\]
Inoltre, la soluzione è data da:
\[
\mathbf{x}^* = (A^\top A)^{-1}A^\top \mathbf{b} = V' \Sigma'^{-1} U'^\top \mathbf{b}.
\]
\end{proposition}
\end{frame}

\section{Dynamic Mode Decomposition - DMD}
\subsection{Definizione}
\begin{frame}{Dynamic Mode Decomposition - DMD}
\begin{itemize}
    \item Dato un insieme di osservazioni spazio-temporali composte da \(m\) parametri in \(n\) istanti di tempo equidistanti \[ \{\mathbf{x}(t_k),\mathbf{x}(t_k')\}_{k=1}^n \subset \R^m \ e \ t_k'=t_k + \Delta t \]  l'obiettivo della DMD è individuare la matrice \(A\) che descriva al meglio l'evoluzione del sistema dinamico che ha prodotto i dati
\end{itemize}

\[\mathbf{x}_{k+1} \simeq A\mathbf{x}_k\]

\end{frame}
\begin{frame}
\begin{itemize}
    \item Le osservazioni possono essere racchiuse in due matrici così costruite
\end{itemize}

\[ X= \begin{bmatrix}
| & | &  & | \\
\mathbf{x}(t_1) & \mathbf{x}(t_2) & \dots & \mathbf{x}(t_n)\\
| & | & & |
\end{bmatrix} \quad 
X'= \begin{bmatrix}
| & | &  & | \\
\mathbf{x}(t_1') & \mathbf{x}(t_2') & \dots & \mathbf{x}(t_n')\\
| & | & & |
\end{bmatrix}
\]
\begin{itemize}
    \item Una volta ottenuta la matrice \(A\) saremo in grado di ricostruire l'evoluzione del sistema a partire dalla condizione iniziale \(\mathbf{x}(t_1)\). \\
    Il dataset ottenuto avrà la seguente formulazione
\end{itemize}

\[
X \simeq
\begin{bmatrix}
| & | & & | & & | \\
\mathbf{x}_1 & A \mathbf{x}_1 & \dots & A^i \mathbf{x}_1 & \dots &  A^{n-1}\mathbf{x}_1 \\
| & | & & | & & | \\
\end{bmatrix}
\]
\end{frame}

\begin{frame}
\begin{itemize}
    \item Il problema dell'individuazione della matrice \(A\) può essere così formulato
    \[A= \min_{M \in \R^{m \times m}}{\|X' -MX\|_F } \]
     e tramite la  \textbf{Proposizione \ref{Minim}} otteniamo la seguente soluzione
    \[ A= X' X^+ \]
\end{itemize}
\end{frame}

\begin{frame}
\begin{itemize}
\item Poiché nelle applicazioni di comune impiego i parametri osservati sono spesso dell'ordine di \( 10^6\) il calcolo della matrice \(A\) risulta essere computazionalmente ingestibile
\item Per ovviare a questa problematica ricorriamo alla\textit{ Proper Orthogonal Decomposition} (\textbf{Teorema \ref{teo_pod}}) che permette di approssimare al meglio lo Span della matrice \(A\) con un minor numero di vettori.
\end{itemize}
\end{frame}

\subsection{Implementazione}
\begin{frame}{Algoritmo di Tu  \begin{small} Ref. [\ref{burton}] \end{small}}
\begin{itemize}
    \item \textbf{Step 1.} Calcoliamo la fattorizzazione SVD ridotta della matrice \(X = U' \Sigma' V'^\top\), dove \(U' \in \R^{m \times r}\), \(\Sigma' \in \R^{r \times r}\), e \(V' \in \R^{n \times r}\). Osserviamo che \(r\) può essere sia il rango della matrice che un valore a nostra discrezione minore di \(Rg(A)\): ridurre le dimensioni risulta cruciale per grandi matrici $X$ con rango non massimo, alleggerendo il costo computazionale. \\Le colonne di $U'$ sono i \textit{Modi} della \textit{Proper Orthogonal Decomposition}.

    \item \textbf{Step 2.} Dal risultato della \textbf{Proposizione \ref{Minim}} sappiamo che \(A = X' X^+ = X' V' \Sigma'^{-1} U'^\top\). Per evitare una matrice \(A\) di dimensioni eccessive proiettiamo \(A\) sui \textit{Modi} di $U'$ ottenendo:
    \[
    \bar{A} = U'^\top A U' = U'^\top X' V' \Sigma'^{-1} \in \R^{r \times r}
    \]

\end{itemize}
\end{frame}

\begin{frame}
\begin{itemize}
    \item \textbf{Step 3.} Calcoliamo la decomposizione spettrale di \(\bar{A}\): 
    \[
    \bar{A} W = W \Lambda
    \]
    dove \(W \in \R^{r \times r}\) contiene gli autovettori e \(\Lambda\) gli autovalori.

    \item \textbf{Step 4.} I \textit{Modi} \(\Phi \in \R^{m \times r}\) vengono ricostruiti come:
    \[
    \Phi = X' V' \Sigma'^{-1} W
    \]
    Abbiamo ricavato gli autovalori di \(A\) senza calcolare esplicitamente la matrice di iterazione ottenendo:
    \[
    A \Phi = \Phi \Lambda
    \]
\end{itemize}
\end{frame}

\begin{frame}{Ricostruzione del Sistema tramite DMD}
\begin{itemize}
    \item La ricostruzione delle osservazioni si basa su \(\mathbf{x}_k = A^{k-1} \mathbf{x}_1\) per \(k = 2, \dots, M\). Tuttavia, possiamo sfruttare i Modi della DMD. 
    \item Con \(A \Phi = \Phi \Lambda\), possiamo esprimere \(A^{k-1}\) come:
    \[
    A^{k-1} = \Phi \Lambda^{k-1} \Phi^+
    \]
    \item La soluzione diventa:
    \[
    \mathbf{x}_k = \Phi \Lambda^{k-1} \Phi^+ \mathbf{x}_1 = \sum_{j=1}^{r} \phi_j \lambda_j^{k-1} b_j
    \]
    dove \(\mathbf{b} = \Phi^+ \mathbf{x}_1= ({W} {\Lambda})^{-1} \bar{\mathbf{x}}_1 \) con \( \mathbf{x}_1= U' \mathbf{\bar{x}}_1 \) e \(\lambda_j\) sono gli autovalori.
\end{itemize}
\end{frame}


\subsection{Applicazione della DMD al modello epidemiologico S.E.I.R.D.}
\begin{frame}{Modello Epidemiologico S.E.I.R.D}
\begin{itemize}
    \item Il modello S.E.I.R.D. (Suscettibili, Esposti, Infetti, Guariti e Deceduti) è un modello epidemiologico spazio-temporale
\end{itemize}
\[s,e,i,r,d \in \mathscr{C}^{2,1}: \ \ \Omega \times [0,T]  \rightarrow \R^+, \ \Omega \subset \R^n. \]
\begin{tiny}
\begin{equation*}
\begin{cases}
\frac{\partial s(\mathbf{x},t)}{\partial t}+ \beta_i \left(1-\frac{A_e}{n_{pop}} \right)s(\mathbf{x},t) \ i(\mathbf{x},t) + \beta_e \left(1-\frac{A_e}{n_{pop}} \right)s(\mathbf{x},t)  \ e(\mathbf{x},t)  - \nabla \cdot (n_{pop} \  \nu_s \nabla s(\mathbf{x},t))=0 \\
\frac{\partial e(\mathbf{x},t)}{\partial t} - \beta_i \left(1-\frac{A_e}{n_{pop}} \right)s(\mathbf{x},t) \ i(\mathbf{x},t) - \beta_e \left(1-\frac{A_e}{n_{pop}} \right)s(\mathbf{x},t) \ e(\mathbf{x},t)  + \left( \alpha + \gamma_e \right)e(\mathbf{x},t) - \\ \hspace{8.7cm} + \nabla \cdot (n_{pop} \  \nu_e \nabla e(\mathbf{x},t))=0 \\
\frac{\partial i(\mathbf{x},t)}{\partial t} - \alpha \  e(\mathbf{x},t) + \left( \gamma_i + \delta \right) i(\mathbf{x},t) - \nabla \cdot \left( n_{pop} \ \nu_i \nabla i(\mathbf{x},t) \right) =0 \\
\frac{ \partial r(\mathbf{x},t)}{ \partial t} - \gamma_e \  e(\mathbf{x},t) - \gamma_i i(\mathbf{x},t) - \nabla \cdot \left(n_{pop} \ \nu_r \nabla i(\mathbf{x},t) \right) = 0 \\
\frac{ \partial d(\mathbf{x},t)}{ \partial t} - \delta i(\mathbf{x},t) =0 \\
\end{cases}
\end{equation*}
\end{tiny}

\begin{itemize}
    \item     La soluzione numerica del sistema produce un dataset ideale per l'applicazione della DMD composto da un elevato numero di osservazioni in istanti di tempo equidistanti fra loro
\end{itemize}
\end{frame}

\begin{frame}{Discretizzazione del Modello }
\begin{itemize}
    \item Consideriamo il modello semplificato analizzando un dominio spaziale monodimensionale, \(\Omega \subset \R  \), e tramite il metodo delle differenze finite all'indietro otteniamo il problema discreto 
\end{itemize}
\begin{tiny}

\begin{equation*}
\begin{cases}
\frac{S_i^{n+1} - S_i^n}{\tau} - n_{pop} \nu_s \frac{S_{i+1}^{n+1} - 2S_i^{n+1} + S_{i-1}^{n+1}}{h^2} =  - \beta_i \left(1 - \frac{A_e}{n_{pop}} \right) S_i^n I_i^n - \beta_e \left(1 - \frac{A_e}{n_{pop}} \right) S_i^n E_i^n = :{f_S}_i^n \\

\frac{E_i^{n+1} - E_i^n}{\tau} - n_{pop} \nu_e \frac{E_{i+1}^{n+1} - 2E_i^{n+1} + E_{i-1}^{n+1}}{h^2} =   \beta_i \left(1 - \frac{A_e}{n_{pop}} \right) S_i^n I_i^n + \beta_e \left(1 - \frac{A_e}{n_{pop}} \right) S_i^n E_i^n = :{f_E}_i^n \\
\frac{I_i^{n+1} - I_i^n}{\tau}  - n_{pop} \nu_i \frac{I_{i+1}^{n+1} - 2I_i^{n+1} + I_{i-1}^{n+1}}{h^2} =   \alpha E_i^n - \left( \gamma_i + \delta \right) I_i^n = :{f_I}_i^n \\
\frac{R_i^{n+1} - R_i^n}{\tau} - n_{pop} \nu_r \frac{R_{i+1}^{n+1} - 2R_i^{n+1} + R_{i-1}^{n+1}}{h^2} =  \gamma_e E_i^n + \gamma_i I_i^n = :{f_R}_i^n \\
\frac{D_i^{n+1} - D_i^n}{\tau} =  \delta I_i^n =:{f_D}_i^n \\
\end{cases}
\end{equation*}
\end{tiny}
\end{frame}

\begin{frame}
\begin{proposition}\label{prop 1}
\textit{Hp:} Data $M\in\mathbb{R}^{m \times n}$\\
\textit{Ts:} $D=M^\top      M$ e $S=MM^\top      $ sono  matrici simmetriche e semi definite positive.
% semidefinite
\end{proposition}

\begin{proposition}\label{prop 2}
\textit{Hp:} Sia $A \in \R^{m \times n}$ con $m \geq n$ e $\{\lambda_i(A^\top A)\}_{i=1}^{n}$ e $\{\lambda_j(A A^\top)\}_{j=1}^{m} $ gli autovalori di ciascuna matrice quadrata \\
\textit{Ts:} $ \{\lambda_i(A^\top A)\}_{i=1}^{n} \subset \{\lambda_j(A A^\top)\}_{j=1}^{m} $ .
\end{proposition}

\begin{corollary}
\textit{Hp:} Sia $M \in \R^{n \times n}$ con autovalori $\mathbf{\lambda}=[\lambda_1, \dots, \lambda_n ]$\\
\textit{Ts:} Gli autovalori di $M^\top      M \ \text{e} \ MM^\top      $ sono gli stessi e valgono $\mathbf{\lambda}^2=[\lambda_1^2, \dots, \lambda_n^2 ]$
\end{corollary}
\end{frame}

\begin{frame}
\begin{proposition}\label{Minim}
\textbf{}\\
\textit{Hp:} Data una matrice \( M \in \R^{m \times n} \) con \( m \geq n \), \(\operatorname*{Rg}(M) = r\) e \(M= U\Sigma V^\top \) la sua decomposizione SVD \\
\textit{Ts:} Allora \(M^+={\operatorname*{argmin}}_{B \in \R^{n \times m}}{\|MB - I_m\|_F }\) dove con \(M^+\) denotiamo la matrice pseudoinversa e grazie alle proprietà della fattorizzazione SVD siamo in grado di trovare la sua formulazione esplicita \( M^+:=V\Sigma^+U^\top \), dove \( \Sigma_{i,j}^+=0 \ \text{se} \ \Sigma_{i,j}=0 \ \text{e} \ \Sigma_{i,j}^+= \frac{1}{\sigma_i} \ \text{se} \  \Sigma_{i,j}=\sigma_i \).
\end{proposition}
\end{frame}

\begin{frame}
\begin{theorem}\label{teo_pod}
\textbf{Singular Value e Proper Orthogonal Decompositions - Ref. [\ref{pod}]}\\
\textit{Hp:} Sia \( Y \in \R^{m \times n}\) una matrice di rango \(d \leq \min{{m,n}}\) e  sia \(Y=U \Sigma V^\top\) la sua scomposizione a valori singolari.\\
\textit{Ts:} Allora \( \forall \ l \in \{1,\dots,d\} \) la soluzione al seguente problema di ottimizzazione vincolata
\[
\max_{\mathbf{\tilde{u}}_1, \dots, \mathbf{\tilde{u}}_l \in \R^m} \sum_{i=1}^l \sum_{j=1}^n | \mathbf{y}_j^\top \mathbf{\tilde{u}}_i |^2 \quad s.t. \quad  \mathbf{\tilde{u}}_i^\top \mathbf{\tilde{u}}_j = \delta_{i,j} \ 1 \leq i,j \leq l
\]
 è data dai vettori singolari \( \{ \mathbf{u}_i\}_{i=1}^l\) ovvero le prime \(l\) colonne di U.
\end{theorem}
\end{frame}
